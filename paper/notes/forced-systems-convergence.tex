These notes develop two complementary local conditions for convergence of the
block Jacobi-Newton method applied to forced discrete Euler-Lagrange equations.
The first (Theorem~\ref{thm:forced-local-implies-global}) gives global
positive-definiteness of $\Sym(H^f)$ from a per-edge symmetric-positive
condition, but is \emph{not} by itself enough to guarantee convergence. The
second (Theorem~\ref{thm:forced-local-dominance}) is a per-edge dominance
condition that does imply $\rho(J^f) < 1$ rigorously, and is the headline result
of these notes.

\section{Setup and notation}

Recall the forced Hessian $H^f$ is block-tridiagonal with diagonal blocks
$\mathcal{D}_k = \mathcal{B}_{k-1} + \mathcal{A}_k$ and off-diagonal blocks
$\mathcal{C}^+_k$ (super-diagonal) and $\mathcal{C}^-_k$ (sub-diagonal), where
\begin{align*}
    \mathcal{A}_k &= \mdif{11} \Ld(q_k, q_{k+1}) + \mdif{1} f_d^-(q_k, q_{k+1}), \\
    \mathcal{B}_k &= \mdif{22} \Ld(q_k, q_{k+1}) + \mdif{2} f_d^+(q_k, q_{k+1}), \\
    \mathcal{C}^+_k &= \mdif{12} \Ld(q_k, q_{k+1}) + \mdif{2} f_d^-(q_k, q_{k+1}), \\
    \mathcal{C}^-_k &= \mdif{21} \Ld(q_k, q_{k+1}) + \mdif{1} f_d^+(q_k, q_{k+1}).
\end{align*}
All four matrices depend only on the edge $(q_k, q_{k+1})$, $k = 0, 1, \dots, N-1$.
The diagonal block at the interior node $j \in \{1, \ldots, N-1\}$ is
$\mathcal{D}_j = \mathcal{A}_j + \mathcal{B}_{j-1}$, mixing the two edges incident
to node $j$. The local Hessian on edge $k$ is
\[
    H_k^f = \begin{pmatrix} \mathcal{A}_k & \mathcal{C}^+_k \\ \mathcal{C}^-_k & \mathcal{B}_k \end{pmatrix} \in \RR^{2n \times 2n}.
\]
Since $H^f$ is generally non-symmetric, ``positive definite'' is interpreted as
$x^T H^f x > 0$ for all $x \neq 0$, equivalently $\Sym(H^f) = \tfrac{1}{2}(H^f + (H^f)^T)$ is
positive definite in the usual sense.

The Jacobi matrix associated with the block-diagonal splitting $H^f = D - (L + U)$
(with $D = \operatorname{diag}(\mathcal{D}_1, \ldots, \mathcal{D}_{N-1})$) is
$J^f = -D^{-1}(L+U) = I - D^{-1} H^f$. Its only nonzero blocks are
\[
    (J^f)_{j, j+1} = -\mathcal{D}_j^{-1}\, \mathcal{C}^+_j, \qquad
    (J^f)_{j, j-1} = -\mathcal{D}_j^{-1}\, \mathcal{C}^-_{j-1},
\]
with the boundary conventions $\mathcal{C}^+_{N-1}\!=\!\mathcal{C}^-_0\!=\!0$
(those blocks do not appear in $H^f$).

\section{Auxiliary fact: positive symmetric part bounds the inverse}

The following standard inequality is used throughout.

\begin{lemma}\label{lem:sym-bound-inverse}
    Let $M \in \RR^{m\times m}$. If $\Sym(M)$ is positive definite, then $M$ is
    invertible and
    \[
        \|M^{-1}\|_2 \;\leq\; \frac{1}{\lambda_{\min}(\Sym(M))}.
    \]
\end{lemma}
\begin{proof}
    For any $v \neq 0$, Cauchy--Schwarz gives
    $\|Mv\|_2\,\|v\|_2 \geq v^T M v = v^T \Sym(M)\, v \geq \lambda_{\min}(\Sym(M))\,\|v\|_2^2$,
    hence $\|Mv\|_2 \geq \lambda_{\min}(\Sym(M))\,\|v\|_2 > 0$. So $M$ is injective
    (hence invertible since $M$ is square), $\sigma_{\min}(M) \geq \lambda_{\min}(\Sym(M))$,
    and $\|M^{-1}\|_2 = 1/\sigma_{\min}(M) \leq 1/\lambda_{\min}(\Sym(M))$.
\end{proof}

\section{Local sym-positivity implies global sym-positivity}

\begin{theorem}[Local sym-positivity implies global]\label{thm:forced-local-implies-global}
    If $\Sym(H_k^f)$ is positive definite for every $k$, then $\Sym(H^f)$ is positive definite,
    and each diagonal block $\mathcal{D}_j$ is invertible.
\end{theorem}
\begin{proof}
    Let $x = (x_1, \ldots, x_{N-1})$ with $x_j \in \RR^n$, extended by $x_0 = x_N = 0$.
    For $k = 0, \ldots, N-1$, define edge vectors $y_k = (x_k,\, x_{k+1})^T \in \RR^{2n}$.
    A direct expansion gives
    \begin{align*}
        \sum_{k=0}^{N-1} y_k^T H_k^f\, y_k
        &= \sum_{k=0}^{N-1} \Bigl(
            x_k^T \mathcal{A}_k x_k
            + x_{k+1}^T \mathcal{B}_k x_{k+1}
            + x_k^T \mathcal{C}^+_k x_{k+1}
            + x_{k+1}^T \mathcal{C}^-_k x_k
           \Bigr) \\
        &= \sum_{k=1}^{N-1} x_k^T (\mathcal{A}_k + \mathcal{B}_{k-1}) x_k
           + \sum_{k=1}^{N-2} \Bigl( x_k^T \mathcal{C}^+_k x_{k+1} + x_{k+1}^T \mathcal{C}^-_k x_k \Bigr) \\
        &= \sum_{k=1}^{N-1} x_k^T \mathcal{D}_k x_k
           + \sum_{k=1}^{N-2} \Bigl( x_k^T \mathcal{C}^+_k x_{k+1} + x_{k+1}^T \mathcal{C}^-_k x_k \Bigr) \\
        &= x^T H^f x.
    \end{align*}
    The second equality uses $x_0 = x_N = 0$ (boundary edge terms collapse to
    pure $\mathcal{B}_0$ and $\mathcal{A}_{N-1}$ pieces, which are absorbed into
    $\mathcal{D}_1$ and $\mathcal{D}_{N-1}$); the third uses
    $\mathcal{D}_k = \mathcal{A}_k + \mathcal{B}_{k-1}$.

    Since $z^T M z = z^T \Sym(M) z$ for any square $M$, the hypothesis
    gives $y_k^T H_k^f y_k > 0$ whenever $y_k \neq 0$. If $x \neq 0$, some
    $x_j \neq 0$, making both $y_{j-1}$ and $y_j$ nonzero, so the sum is strictly
    positive. Hence $x^T H^f x > 0$ for all $x \neq 0$, i.e., $\Sym(H^f) > 0$.
    Invertibility of $\mathcal{D}_j$ follows from Lemma~\ref{lem:sym-bound-inverse}:
    restricting $x$ to its $j$-th block component gives $v^T \mathcal{D}_j v > 0$
    for all $v \neq 0$, so $\Sym(\mathcal{D}_j) > 0$.
\end{proof}

\section{The gap: $\Sym(H^f) > 0$ does not imply $\rho(J^f) < 1$}\label{sec:gap}

To complete the convergence argument from local data, one would need:

\medskip
\textbf{Desired implication.} \textit{If $\Sym(H^f)$ is positive definite, then $\rho(J^f) < 1$.}
\medskip

This implication is \textbf{false} in general. A minimal counter-example:
\[
    M = \begin{pmatrix} 1 & a \\ -a & 1 \end{pmatrix},
    \quad D = I,
    \quad J = I - D^{-1}M = \begin{pmatrix} 0 & -a \\ a & 0 \end{pmatrix}.
\]
Here $\Sym(M) = I$ is positive definite for any $a$, but $\rho(J) = |a|$,
which is unbounded. The Jacobi iteration couples the off-diagonal
block of $M$ back into the iteration; if that block is large and skew-symmetric, the
iteration spirals out even though the symmetric part stays positive.

In the unforced (symmetric) case $\mathcal{C}^+_k = (\mathcal{C}^-_k)^T$, so the skew
part of $H$ vanishes and this failure mode is excluded. In the forced case it can arise
whenever $\mathcal{C}^+_k \neq (\mathcal{C}^-_k)^T$ with large asymmetry.

\subsection*{Why $\lambda_{\min}(\Sym(M)) > \|\Skew(M)\|_2$ is the natural threshold}

Before stating the conjecture, it is worth justifying why this specific
combination of $\Sym$ and $\Skew$ is the natural way to make ``$M$ is comfortably
positive'' precise for non-symmetric $M$. There are three independent reasons,
each pointing to the same threshold.

\paragraph{(i) It reduces to positive-definiteness in the symmetric case.}
If $\Skew(M) = 0$, the condition is $\lambda_{\min}(\Sym(M)) > 0$, i.e., the
ordinary $M \succ 0$. So the condition is a one-parameter quantitative
extension of positive-definiteness: take a positive-definite symmetric matrix
and allow a skew perturbation of size strictly smaller than its smallest
eigenvalue. Any other generalization to the non-symmetric setting must either
match this reduction (and so look essentially the same) or be a different
strengthening (such as block-diagonal dominance).

\paragraph{(ii) It is the tight threshold for the minimal counter-example.}
For the example
$M = \bigl(\begin{smallmatrix} 1 & a \\ -a & 1 \end{smallmatrix}\bigr)$ above,
$\lambda_{\min}(\Sym(M)) = 1$ and $\|\Skew(M)\|_2 = |a|$, and the Jacobi
spectral radius is exactly $|a|$. The condition
$\lambda_{\min}(\Sym(M)) > \|\Skew(M)\|_2$ becomes $1 > |a|$, which is
\emph{exactly} the convergence threshold $\rho(J) < 1$. The condition is not
just sufficient for this example; it is sharp. Any weaker condition (i.e.\
allowing more skew relative to symmetric part) would fail to rule out the
non-convergent regime.

\paragraph{(iii) It places the spectrum of $M$ in a sector of the right half-plane.}
This is the geometric reason. For any complex eigenvector $v$ of a real matrix $M$,
writing $\lambda = \alpha + i\beta$ for the corresponding eigenvalue,
\[
    \alpha = \frac{\operatorname{Re}(v^* M v)}{\|v\|^2}
           = \frac{v^* \Sym(M)\, v}{\|v\|^2} \in [\lambda_{\min}(\Sym(M)), \lambda_{\max}(\Sym(M))],
\]
\[
    |\beta| = \frac{|\operatorname{Im}(v^* M v)|}{\|v\|^2}
            = \frac{|v^* \Skew(M)\, v|}{\|v\|^2} \leq \|\Skew(M)\|_2.
\]
(The first follows because $v^* \Sym(M) v$ is real and equal to a Rayleigh
quotient on the real symmetric matrix $\Sym(M)$. The second follows because
$v^* \Skew(M) v$ is purely imaginary --- the conjugate of $v^* \Skew(M) v$ is
$v^* \Skew(M)^T v = -v^* \Skew(M) v$ --- and its magnitude is bounded by
$\|\Skew(M)\|_2$ via the standard inequality
$|v^* A v| \leq \|A\|_2 \|v\|^2$ for any matrix $A$.)

So the spectrum of $M$ is contained in the rectangle
\[
    R := [\lambda_{\min}(\Sym(M)), \lambda_{\max}(\Sym(M))]
         \times i[-\|\Skew(M)\|_2, \|\Skew(M)\|_2] \subset \mathbb{C}.
\]
The condition $\lambda_{\min}(\Sym(M)) > \|\Skew(M)\|_2$ shoves $R$ strictly
into the open wedge $W := \{z \in \mathbb{C} : \operatorname{Re}(z) > |\operatorname{Im}(z)|\}$,
the $45^\circ$ sector strictly inside the right half-plane:
\[
    \begin{array}{c}
    \text{spectrum of } M \subset R \subset W \\[0.2em]
    \text{i.e., every eigenvalue } \lambda \text{ of } M
    \text{ satisfies } \operatorname{Re}(\lambda) > |\operatorname{Im}(\lambda)| > 0.
    \end{array}
\]
This is the property of being \emph{sectorial} (or \emph{uniformly accretive}):
$M$ has its entire spectrum in a sector around the positive real axis,
bounded away from the imaginary axis. It is the cleanest geometric way to
say ``$M$ is a positive-definite matrix, plus a not-too-large rotation''.

\paragraph{Why sectorial $M$ is a natural hypothesis for $\rho(J) < 1$.}
The Jacobi iteration matrix is $J = I - D^{-1}M$, so $\rho(J) < 1$ corresponds
to the spectrum of $D^{-1}M$ lying inside the open disc of radius $1$ centered
at $1$. If $D = c I$ for some $c > 0$ (the case in the minimal counter-example),
then $D^{-1}M = c^{-1}M$ inherits the sectorial location of $M$ scaled by
$1/c$, and the condition $\lambda_{\min}(\Sym(M)) > \|\Skew(M)\|_2$ exactly
matches the scaling at which the sector fits inside the disc-of-radius-$1$-at-$1$
(see Appendix~\ref{app:conjecture-scalar}). For general block-diagonal $D$,
$D^{-1}M$ is a matrix-structured combination of $D^{-1}$, $\Sym(M)$ and
$\Skew(M)$, and the sectorial property of $M$ alone does not transfer
mechanically to $D^{-1}M$. This is the reason the conjecture is hard in the
general block case: the geometric heuristic (i)--(iii) suggests
$\lambda_{\min}(\Sym(M)) > \|\Skew(M)\|_2$ is the right threshold, but the
algebraic bridge from $M$ to $D^{-1}M$ does not preserve the sectorial location
exactly.

\subsection*{The conjecture}

With this motivation:

\medskip
\textbf{Working conjecture.}
\textit{If $\lambda_{\min}(\Sym(H^f)) > \|\Skew(H^f)\|_2$, then $\rho(J^f) < 1$.}
\medskip

In the scalar $2 \times 2$ block case ($n = 1$, $N = 3$) it is provable
by direct computation (see Appendix~\ref{app:conjecture-scalar}). The general
block case remains open: standard tools (Lyapunov, field-of-values, $H$-matrix
theory) all stop short of supplying it cleanly.

\section{A different local condition that does imply convergence}\label{sec:local-dominance}

Rather than trying to close the gap from $\Sym(H^f) > 0$, we strengthen the
per-edge condition itself. The strengthening replaces ``$H_k^f$ is sym-positive''
with a quantitative dominance of the symmetric part of the diagonal sub-blocks
of $H_k^f$ over its off-diagonal sub-blocks. This is sufficient for $\rho(J^f) < 1$
\emph{directly}, without going through global $\Sym(H^f) > 0$.

\begin{theorem}[Local edge dominance implies $\rho(J^f) < 1$]\label{thm:forced-local-dominance}
    Suppose that for every edge $k \in \{0, 1, \ldots, N-1\}$,
    \begin{equation}
        \lambda_{\min}(\Sym(\mathcal{A}_k)) > \|\mathcal{C}^+_k\|_2,
        \qquad
        \lambda_{\min}(\Sym(\mathcal{B}_k)) > \|\mathcal{C}^-_k\|_2,
        \label{eq:local-edge-dominance-notes}
    \end{equation}
    with the boundary convention $\mathcal{C}^+_{N-1}\!=\!\mathcal{C}^-_0\!=\!0$
    (so that for $k = N-1$ only the first inequality is restrictive and for
    $k = 0$ only the second). Then:
    \begin{enumerate}
        \item every diagonal block $\mathcal{D}_j$ ($j = 1, \ldots, N-1$) is invertible;
        \item $\rho(J^f) \leq \|J^f\|_{\infty, 2} < 1$, where $\|\cdot\|_{\infty, 2}$
              is the operator norm induced by the block max-norm
              $\|x\|_{\infty, 2} = \max_j \|x_j\|_2$;
        \item by~\cite[Lemma~4.4]{ferraroParallelIterativeMethod2025}, the forced
              Jacobi-Newton method converges locally at $\mathbf{q}^*$.
    \end{enumerate}
\end{theorem}
\begin{proof}
    \emph{Invertibility of $\mathcal{D}_j$.} By Weyl's inequality applied to the
    symmetric parts (which are real symmetric matrices, so $\lambda_{\min}$ is
    superadditive),
    \[
        \lambda_{\min}(\Sym(\mathcal{D}_j))
        = \lambda_{\min}(\Sym(\mathcal{A}_j) + \Sym(\mathcal{B}_{j-1}))
        \geq \lambda_{\min}(\Sym(\mathcal{A}_j)) + \lambda_{\min}(\Sym(\mathcal{B}_{j-1})).
    \]
    By hypothesis~\eqref{eq:local-edge-dominance-notes} on edges $j$ and $j-1$,
    the right-hand side is strictly greater than
    $\|\mathcal{C}^+_j\|_2 + \|\mathcal{C}^-_{j-1}\|_2 \geq 0$, so
    $\Sym(\mathcal{D}_j) \succ 0$. Lemma~\ref{lem:sym-bound-inverse} gives
    invertibility of $\mathcal{D}_j$ and
    \begin{equation}
        \|\mathcal{D}_j^{-1}\|_2 \;\leq\; \frac{1}{\lambda_{\min}(\Sym(\mathcal{D}_j))}.
        \label{eq:Dinv-bound}
    \end{equation}

    \emph{Bound on the Jacobi norm.} Equip $\RR^{(N-1)n}$, decomposed into
    interior-node blocks of size $n$, with the block max-norm
    $\|x\|_{\infty, 2} = \max_j \|x_j\|_2$. For any $x$ and any $j$,
    \[
        (J^f x)_j = -\mathcal{D}_j^{-1} \mathcal{C}^-_{j-1}\, x_{j-1}
                    - \mathcal{D}_j^{-1} \mathcal{C}^+_j\, x_{j+1}
    \]
    (using the boundary convention for $j = 1$ and $j = N-1$). Hence
    \begin{align*}
        \|(J^f x)_j\|_2
        &\leq \|\mathcal{D}_j^{-1} \mathcal{C}^-_{j-1}\|_2 \|x_{j-1}\|_2
             + \|\mathcal{D}_j^{-1} \mathcal{C}^+_j\|_2 \|x_{j+1}\|_2 \\
        &\leq \bigl(\|\mathcal{D}_j^{-1} \mathcal{C}^-_{j-1}\|_2 + \|\mathcal{D}_j^{-1} \mathcal{C}^+_j\|_2\bigr) \, \|x\|_{\infty, 2}.
    \end{align*}
    Taking $\max_j$, the induced operator norm satisfies
    \[
        \|J^f\|_{\infty, 2}
        \;\leq\; \max_{j=1, \ldots, N-1}
            \bigl( \|\mathcal{D}_j^{-1} \mathcal{C}^-_{j-1}\|_2
                  + \|\mathcal{D}_j^{-1} \mathcal{C}^+_j\|_2 \bigr).
    \]
    Using submultiplicativity of the spectral norm and~\eqref{eq:Dinv-bound},
    \begin{align*}
        \|\mathcal{D}_j^{-1} \mathcal{C}^-_{j-1}\|_2 + \|\mathcal{D}_j^{-1} \mathcal{C}^+_j\|_2
        &\leq \|\mathcal{D}_j^{-1}\|_2\bigl(\|\mathcal{C}^-_{j-1}\|_2 + \|\mathcal{C}^+_j\|_2\bigr) \\
        &\leq \frac{\|\mathcal{C}^-_{j-1}\|_2 + \|\mathcal{C}^+_j\|_2}{\lambda_{\min}(\Sym(\mathcal{A}_j)) + \lambda_{\min}(\Sym(\mathcal{B}_{j-1}))} \\
        &< 1,
    \end{align*}
    the last inequality by adding~\eqref{eq:local-edge-dominance-notes} for edge $j$
    (the $\mathcal{A}_j$ inequality) and for edge $j-1$ (the $\mathcal{B}_{j-1}$
    inequality). Therefore $\|J^f\|_{\infty, 2} < 1$, and the bound is uniform in $j$.

    \emph{Spectral radius.} For any induced operator norm $\|\cdot\|$, the bound
    $\rho(J^f) \leq \|J^f\|$ holds; with $\|\cdot\| = \|\cdot\|_{\infty, 2}$,
    we obtain $\rho(J^f) < 1$.

    \emph{Convergence.} Both hypotheses of~\cite[Lemma~4.4]{ferraroParallelIterativeMethod2025}
    --- invertibility of each $\mathcal{D}_j$ and $\rho(J^f) < 1$ --- are met,
    yielding local convergence of the forced Jacobi-Newton iteration to
    $\mathbf{q}^*$.
\end{proof}

\section{Comparison of the two local conditions}\label{sec:comparison}

The two local conditions are not nested:

\begin{itemize}
    \item Theorem~\ref{thm:forced-local-implies-global} requires
          $\Sym(H_k^f) \succ 0$. This implies $\Sym(\mathcal{A}_k), \Sym(\mathcal{B}_k) \succ 0$
          and a Schur-complement bound mixing $\mathcal{C}^+_k$ and $\mathcal{C}^-_k$, but does
          \emph{not} imply~\eqref{eq:local-edge-dominance-notes}.

    \item Theorem~\ref{thm:forced-local-dominance} requires the edge-dominance
          inequalities~\eqref{eq:local-edge-dominance-notes}, which similarly do
          \emph{not} imply $\Sym(H_k^f) \succ 0$ (the off-diagonal sub-blocks
          can combine into a large indefinite symmetric part).
\end{itemize}

The pertinent counter-examples (taking $n = 1$ for transparency):

\begin{enumerate}
    \item \emph{Sym-positive but not edge-dominant.} $\mathcal{A}_k = \mathcal{B}_k = 1$,
          $\mathcal{C}^+_k = 2$, $\mathcal{C}^-_k = -2$. Then
          $H_k^f = \bigl(\begin{smallmatrix}1 & 2 \\ -2 & 1\end{smallmatrix}\bigr)$,
          $\Sym(H_k^f) = I \succ 0$, but $\lambda_{\min}(\Sym(\mathcal{A}_k)) = 1 < 2 = \|\mathcal{C}^+_k\|_2$.

    \item \emph{Edge-dominant but not sym-positive.}
          $\mathcal{A}_k = 2.1$, $\mathcal{B}_k = 0.1$, $\mathcal{C}^+_k = 2$, $\mathcal{C}^-_k = 0.05$.
          Edge dominance: $2.1 > 2$ and $0.1 > 0.05$, both hold.
          $\Sym(H_k^f) = \bigl(\begin{smallmatrix}2.1 & 1.025 \\ 1.025 & 0.1\end{smallmatrix}\bigr)$
          has determinant $0.21 - 1.0506 < 0$, hence is \emph{not} positive definite.
\end{enumerate}

Both Theorem~\ref{thm:forced-local-implies-global} and~\ref{thm:forced-local-dominance} are
genuine, complementary, and neither subsumes the other.

In practice, Theorem~\ref{thm:forced-local-dominance} is what is checked when
one wants a rigorous \emph{a-priori} certificate of convergence, since it is the
one that closes the loop to $\rho(J^f) < 1$. Theorem~\ref{thm:forced-local-implies-global}
remains useful as a structural fact about $H^f$ (positive symmetric part) and as
a stepping stone in case the conjecture of Section~\ref{sec:gap} is ever proven.

\section{Edge-dominance in the symmetric and unforced cases}\label{sec:recovers}

\begin{remark}[Unforced case]\label{rem:unforced-recovery}
    With $f_d^\pm \equiv 0$, we have $\mathcal{A}_k = \mdif{11} \Ld$ and
    $\mathcal{C}^+_k = \mdif{12} \Ld$, $\mathcal{B}_k = \mdif{22}\Ld$,
    $\mathcal{C}^-_k = \mdif{21}\Ld = (\mdif{12}\Ld)^T$. All blocks are
    symmetric or transposes of one another, and the dominance
    condition~\eqref{eq:local-edge-dominance-notes} becomes
    \[
        \lambda_{\min}(\mdif{11}\Ld(q_k, q_{k+1})) > \|\mdif{12}\Ld(q_k, q_{k+1})\|_2,
        \qquad
        \lambda_{\min}(\mdif{22}\Ld(q_k, q_{k+1})) > \|\mdif{21}\Ld(q_k, q_{k+1})\|_2.
    \]
    This is a strict diagonal-dominance condition on the second derivatives of
    $\Ld$, sufficient for the unforced parallel method to converge.
\end{remark}

\begin{remark}[Symmetric forces case]\label{rem:sym-recovery}
    Under the symmetric force condition $(\mdif{2} f_d^-)^T = \mdif{1} f_d^+$
    of the main text, we have $\mathcal{C}^-_k = (\mathcal{C}^+_k)^T$, so
    $\|\mathcal{C}^+_k\|_2 = \|\mathcal{C}^-_k\|_2$. The edge-dominance
    inequalities~\eqref{eq:local-edge-dominance-notes} collapse to
    $\min(\lambda_{\min}(\Sym(\mathcal{A}_k)), \lambda_{\min}(\Sym(\mathcal{B}_k))) > \|\mathcal{C}^+_k\|_2$.
\end{remark}

\section{Practical checkability and time complexity}\label{sec:practical}

Theorem~\ref{thm:forced-local-dominance} reduces the convergence test to
$N$ independent edge tests, each involving:
\begin{itemize}
    \item one smallest-eigenvalue computation on $\Sym(\mathcal{A}_k)$ and one
          on $\Sym(\mathcal{B}_k)$ ($n \times n$ symmetric), cost $O(n^3)$ per matrix;
    \item one spectral-norm computation on $\mathcal{C}^+_k$ and one on
          $\mathcal{C}^-_k$ ($n \times n$ general), cost $O(n^3)$ via SVD
          (cheaper via power iteration on $\mathcal{C}^T \mathcal{C}$ if only a
          sharp upper bound is needed).
\end{itemize}
The total serial cost is $O(N n^3)$, trivially parallelizable across edges to
$O(n^3)$ on $N$ processors. This is the same order as the unforced check and
strictly cheaper than the $O((Nn)^3)$ cost of computing $\rho(J^f)$ or the
eigendecomposition of $\Sym(H^f)$ directly.

Compared with Theorem~\ref{thm:forced-local-implies-global}, the per-edge work
is comparable: there one tests $\Sym(H_k^f) \succ 0$, a single $2n \times 2n$
symmetric-positive-definiteness check at $O(n^3)$. The edge-dominance test
of~\eqref{eq:local-edge-dominance-notes} replaces this with two $n \times n$
eigenvalue computations plus two spectral-norm computations: same complexity
class.

\section{Where the perturbation route fits in}

Theorem~\ref{thm:forced-local-dominance} is genuinely local in the (sub)blocks
$\mathcal{A}_k, \mathcal{B}_k, \mathcal{C}^\pm_k$, but it does require enough
``regularity'' of $\mathcal{A}_k$ and $\mathcal{B}_k$ relative to the cross
blocks. If~\eqref{eq:local-edge-dominance-notes} happens to fail but the
unforced problem (with the same discrete Lagrangian) is known to converge, the
perturbation condition of the main text
(Theorem~\ref{thm:forced-perturbation} / Remark~\ref{rem:forced-perturbation-checkable})
applies: small forces relative to the unforced contraction margin still give
convergence. The two results are independent and cover different regimes:
edge-dominance is a one-shot ``the forced problem is good on its own'' test,
whereas the perturbation route says ``the forced problem inherits convergence
from a known unforced problem''.

\section{The working conjecture in the scalar 2-node case}\label{app:conjecture-scalar}

We verify the conjecture of Section~\ref{sec:gap}
($\lambda_{\min}(\Sym(H^f)) > \|\Skew(H^f)\|_2 \Rightarrow \rho(J^f) < 1$)
for $n = 1$ and $N = 3$ (one interior node, $H^f$ a $2\times 2$ matrix).

\noindent\textbf{Setup.} Write $H^f = \bigl(\begin{smallmatrix} d_1 & c^+ \\ c^- & d_2 \end{smallmatrix}\bigr)$
with $d_1, d_2 > 0$. The Jacobi matrix is
\[
    J^f = -\operatorname{diag}(d_1, d_2)^{-1}\bigl(H^f - \operatorname{diag}(d_1, d_2)\bigr)
    = \begin{pmatrix} 0 & -c^+/d_1 \\ -c^-/d_2 & 0 \end{pmatrix},
\]
with characteristic polynomial $\lambda^2 - c^+ c^- / (d_1 d_2)$. Hence
$\rho(J^f)^2 = |c^+ c^-| / (d_1 d_2)$.

Define $s := (c^+ + c^-)/2$ and $a := (c^+ - c^-)/2$, so $c^+ c^- = s^2 - a^2$.
Then
\[
    \Sym(H^f) = \begin{pmatrix} d_1 & s \\ s & d_2 \end{pmatrix},
    \qquad
    \|\Skew(H^f)\|_2 = |a|.
\]
By direct computation
$\lambda_{\min}(\Sym(H^f)) = \tfrac{d_1 + d_2}{2} - \sqrt{\bigl(\tfrac{d_1 - d_2}{2}\bigr)^2 + s^2}$.
The conjecture's hypothesis is therefore
\begin{equation}
    \tfrac{d_1 + d_2}{2} - |a| \;>\; \sqrt{\bigl(\tfrac{d_1 - d_2}{2}\bigr)^2 + s^2}.
    \label{eq:hyp-scalar}
\end{equation}
Both sides being positive (the LHS being positive is what makes the conjecture
non-vacuous), squaring and using $(d_1+d_2)^2 - (d_1-d_2)^2 = 4 d_1 d_2$ rearranges to
\begin{equation}
    d_1 d_2 - (d_1 + d_2)|a| + a^2 \;>\; s^2.
    \label{eq:hyp-scalar-simplified}
\end{equation}

\noindent\textbf{Conclusion.} By AM-GM, $d_1 + d_2 \geq 2 \sqrt{d_1 d_2}$, so
\eqref{eq:hyp-scalar-simplified} implies
\[
    d_1 d_2 - 2 \sqrt{d_1 d_2}\,|a| + a^2 \;\geq\; d_1 d_2 - (d_1 + d_2)|a| + a^2 \;>\; s^2,
\]
i.e., $(\sqrt{d_1 d_2} - |a|)^2 > s^2$ and so $\sqrt{d_1 d_2} > |s| + |a|$
(the LHS of~\eqref{eq:hyp-scalar} being positive forces $\sqrt{d_1 d_2} > |a|$,
so we can take the positive root). Therefore
\[
    |c^+ c^-| \;=\; |s^2 - a^2| \;=\; |s - a|\,|s + a| \;\leq\; (|s| + |a|)^2 \;<\; d_1 d_2,
\]
which gives $\rho(J^f)^2 = |c^+ c^-|/(d_1 d_2) < 1$, as required. \qed

\medskip
The argument above uses the scalar block structure in three essential ways:
(i) $c^+ c^- = s^2 - a^2$ is a single number, (ii) AM-GM on the pair
$(d_1, d_2)$, and (iii) the explicit closed-form formula for
$\lambda_{\min}(\Sym(H^f))$. None of these survives the lift to $n > 1$ blocks
without auxiliary assumptions, which is why the general conjecture remains open.

\section{Conclusion}

The rigorous results in these notes are:
\begin{itemize}
    \item Local sym-pd of $H_k^f$ $\Rightarrow$ global sym-pd of $H^f$ + each $\mathcal{D}_j$ invertible
          (Theorem~\ref{thm:forced-local-implies-global}). \emph{Structural, not sufficient for convergence.}
    \item Local edge dominance~\eqref{eq:local-edge-dominance-notes} $\Rightarrow$ each $\mathcal{D}_j$
          invertible + $\rho(J^f) < 1$ + local convergence
          (Theorem~\ref{thm:forced-local-dominance}). \emph{Rigorous a-priori convergence certificate.}
    \item In the scalar $n=1$, $N=3$ case, the working conjecture
          ``$\lambda_{\min}(\Sym(H^f)) > \|\Skew(H^f)\|_2 \Rightarrow \rho(J^f) < 1$''
          holds (Appendix~\ref{app:conjecture-scalar}).
    \item In the general block case, the working conjecture remains \textbf{open};
          $\Sym(H^f) > 0$ alone is not sufficient (Section~\ref{sec:gap}).
\end{itemize}

The perturbation result in the main text gives a complementary convergence
criterion: small forces around a converging unforced problem still converge.
Together, the edge-dominance theorem and the perturbation theorem cover the
practical regimes of interest.
